    def take_item(self, item_name):
        room_name = self.state['room_name']
        room = self.rooms[room_name]

        # check if the item is in the room
        if item_name not in room['items']:
            return f"There is no {item_name} in this room."

        # add the item to the inventory and remove it from the room
        self.state['inventory'].append(item_name)
        room['items'].remove(item_name)

        return f"You take the {item_name}."
