import threading
import time

class MyClass(threading.Thread):
    def __init__(self):
        threading.Thread.__init__(self)
        self.running = False
        self.start()

    def run(self):
        self.running = True
        while self.running:
            print("Running")
            time.sleep(1)

    def stop(self):
        self.running = False
