 "If the function does not return any data, it returns NULL."

So, just return NULL instead of 0 in your function:

\begin{code}
DELIMITER //

CREATE FUNCTION get_customer_level(
    p_credit DECIMAL(10,2)
)
RETURNS VARCHAR(20)
DETERMINISTIC
BEGIN
    DECLARE lvl VARCHAR(20);

    IF p_credit > 50000 THEN
        SET lvl = 'PLATINUM';
    ELSEIF (p_credit >= 50000 AND p_credit <= 10000) THEN
        SET lvl = 'GOLD';
    ELSEIF p_credit < 10000 THEN
        SET lvl = 'SILVER';
    END IF;

    RETURN lvl;
END //

DELIMITER ;
\end{code}
